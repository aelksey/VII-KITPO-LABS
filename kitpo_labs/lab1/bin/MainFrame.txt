package ru.nstu.app.view;

import javax.swing.*;
import java.awt.*;
import java.io.IOException;
import java.util.List;

import ru.nstu.app.core.CustomList;
import ru.nstu.app.types.UserFactory;
import ru.nstu.app.types.UserType;
import ru.nstu.app.sort.SortFactory;
import ru.nstu.app.sort.SortStrategy;
import ru.nstu.app.serialize.SerializeFactory;
import ru.nstu.app.serialize.SerializeStrategy;

public class MainFrame extends JFrame {

    // Ссылки на текущие абстрактные компоненты ядра (работаем через Object)
    private UserType<Object> currentBuilder;
    private CustomList<Object> currentList;

    // Компоненты интерфейса
    private final JComboBox<String> typeComboBox;
    private final JComboBox<String> sortComboBox;
    private final JComboBox<String> serializeComboBox;
    
    private final DefaultListModel<String> listModel;
    private final JList<String> displayList;
    
    private final JTextField inputField;
    private final JTextField indexField;
    private final JTextField fileField;

    @SuppressWarnings("unchecked")
    public MainFrame() {
        // Настройка главного окна
        setTitle("Лабораторная работа №1 — Вариант 8 (Массив ссылок в элементе)");
        setSize(850, 550);
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        setLocationRelativeTo(null);
        setLayout(new BorderLayout(10, 10));

        // Инициализация базового списка
        currentList = new CustomList<>();

        // --- ВЕРХНЯЯ ПАНЕЛЬ: Управление фабриками ---
        JPanel topPanel = new JPanel(new GridLayout(1, 3, 10, 10));
        topPanel.setBorder(BorderFactory.createTitledBorder("Конфигурация модулей (Фабрики)"));

        // 1. Выбор Типа Данных
        JPanel p1 = new JPanel(new BorderLayout());
        p1.add(new JLabel("Тип данных:"), BorderLayout.NORTH);
        typeComboBox = new JComboBox<>(UserFactory.getTypeNameList().toArray(new String[0]));
        p1.add(typeComboBox, BorderLayout.CENTER);
        topPanel.add(p1);

        // 2. Выбор Алгоритма Сортировки
        JPanel p2 = new JPanel(new BorderLayout());
        p2.add(new JLabel("Алгоритм сортировки:"), BorderLayout.NORTH);
        sortComboBox = new JComboBox<>(SortFactory.getStrategyNameList().toArray(new String[0]));
        p2.add(sortComboBox, BorderLayout.CENTER);
        topPanel.add(p2);

        // 3. Выбор Формата Сериализации
        JPanel p3 = new JPanel(new BorderLayout());
        p3.add(new JLabel("Формат файлов:"), BorderLayout.NORTH);
        serializeComboBox = new JComboBox<>(SerializeFactory.getFormatNameList().toArray(new String[0]));
        p3.add(serializeComboBox, BorderLayout.CENTER);
        topPanel.add(p3);

        add(topPanel, BorderLayout.NORTH);

        // --- ЦЕНТРАЛЬНАЯ ПАНЕЛЬ: Отображение структуры данных ---
        listModel = new DefaultListModel<>();
        displayList = new JList<>(listModel);
        displayList.setFont(new Font("Monospaced", Font.PLAIN, 13));
        JScrollPane scrollPane = new JScrollPane(displayList);
        scrollPane.setBorder(BorderFactory.createTitledBorder("Содержимое CustomList (Вариант 8)"));
        add(scrollPane, BorderLayout.CENTER);

        // --- БОКОВАЯ ПАНЕЛЬ: Операции над списком ---
        JPanel sidePanel = new JPanel();
        sidePanel.setLayout(new BoxLayout(sidePanel, BoxLayout.Y_AXIS));
        sidePanel.setBorder(BorderFactory.createTitledBorder("Операции"));

        // Поля ввода
        inputField = new JTextField(15);
        indexField = new JTextField(5);
        fileField = new JTextField("data_storage", 12);

        // Настройка элементов боковой панели (с отступами)
        addSideComponent(sidePanel, new JLabel("Данные (например, '3.0 4.0' или '42'):"));
        addSideComponent(sidePanel, inputField);
        addSideComponent(sidePanel, new JLabel("Логический индекс:"));
        addSideComponent(sidePanel, indexField);

        JSeparator sep1 = new JSeparator();
        addSideComponent(sidePanel, sep1);

        // Кнопки действий
        JButton addButton = new JButton("Добавить в конец");
        JButton insertButton = new JButton("Вставить по индексу");
        JButton removeButton = new JButton("Удалить по индексу");
        JButton sortButton = new JButton("Выполнить сортировку");

        addSideComponent(sidePanel, addButton);
        addSideComponent(sidePanel, insertButton);
        addSideComponent(sidePanel, removeButton);
        addSideComponent(sidePanel, sortButton);

        JSeparator sep2 = new JSeparator();
        addSideComponent(sidePanel, sep2);

        addSideComponent(sidePanel, new JLabel("Имя файла (без расширения):"));
        addSideComponent(sidePanel, fileField);

        JButton saveButton = new JButton("Сохранить в файл");
        JButton loadButton = new JButton("Загрузить из файла");
        JButton clearButton = new JButton("Очистить список");

        addSideComponent(sidePanel, saveButton);
        addSideComponent(sidePanel, loadButton);
        addSideComponent(sidePanel, clearButton);

        add(sidePanel, BorderLayout.EAST);

        // --- ЛОГИКА И СЛУШАТЕЛИ СОБЫТИЙ ---

        // Первичная инициализация билдера при старте
        currentBuilder = (UserType<Object>) UserFactory.getBuilderByName((String) typeComboBox.getSelectedItem());

        // Слушатель смены типа данных (Смена «на лету» с очисткой старого списка)
        typeComboBox.addActionListener(e -> {
            String selectedType = (String) typeComboBox.getSelectedItem();
            currentBuilder = (UserType<Object>) UserFactory.getBuilderByName(selectedType);
            currentList.clear();
            updateUIList();
            JOptionPane.showMessageDialog(this, "Тип данных переключен на: " + selectedType + ".\nСписок очищен.");
        });

        // Кнопка: Добавить в конец
        addButton.addActionListener(e -> {
            try {
                Object obj = currentBuilder.parseValue(inputField.getText());
                currentList.add(obj);
                updateUIList();
                inputField.setText("");
            } catch (Exception ex) {
                showError("Ошибка парсинга данных. Проверьте формат ввода!");
            }
        });

        // Кнопка: Вставить по индексу
        insertButton.addActionListener(e -> {
            try {
                int idx = Integer.parseInt(indexField.getText());
                Object obj = currentBuilder.parseValue(inputField.getText());
                currentList.insert(idx, obj);
                updateUIList();
                inputField.setText("");
                indexField.setText("");
            } catch (IndexOutOfBoundsException ex) {
                showError("Индекс вышел за пределы размера списка!");
            } catch (Exception ex) {
                showError("Неверный формат индекса или вводимых данных!");
            }
        });

        // Кнопка: Удалить по индексу
        removeButton.addActionListener(e -> {
            try {
                int idx = Integer.parseInt(indexField.getText());
                currentList.remove(idx);
                updateUIList();
                indexField.setText("");
            } catch (IndexOutOfBoundsException ex) {
                showError("Элемент с таким индексом не существует!");
            } catch (Exception ex) {
                showError("Введите корректный числовой индекс!");
            }
        });

        // Кнопка: Сортировать (Стратегия меняется на лету)
        sortButton.addActionListener(e -> {
            String selectedSortName = (String) sortComboBox.getSelectedItem();
            SortStrategy strategy = SortFactory.getStrategyByName(selectedSortName);
            
            currentList.sort(strategy, currentBuilder);
            updateUIList();
            JOptionPane.showMessageDialog(this, "Список отсортирован методом: " + selectedSortName);
        });

        // Кнопка: Сохранить в файл (Стратегия сериализации меняется на лету)
        saveButton.addActionListener(e -> {
            String selectedFormat = (String) serializeComboBox.getSelectedItem();
            SerializeStrategy strategy = SerializeFactory.getStrategyByName(selectedFormat);
            String filename = fileField.getText().trim();

            if (filename.isEmpty()) {
                showError("Введите имя файла!");
                return;
            }

            try {
                strategy.save(filename, currentList, currentBuilder);
                JOptionPane.showMessageDialog(this, "Данные успешно сохранены в формат:\n" + selectedFormat);
            } catch (IOException ex) {
                showError("Ошибка записи в файл: " + ex.getMessage());
            }
        });

        // Кнопка: Загрузить из файла
        loadButton.addActionListener(e -> {
            String selectedFormat = (String) serializeComboBox.getSelectedItem();
            SerializeStrategy strategy = SerializeFactory.getStrategyByName(selectedFormat);
            String filename = fileField.getText().trim();

            if (filename.isEmpty()) {
                showError("Введите имя файла!");
                return;
            }

            try {
                currentList.clear(); // Очищаем текущее состояние структуры данных
                strategy.load(filename, currentList, currentBuilder);
                updateUIList();
                JOptionPane.showMessageDialog(this, "Данные успешно загружены из формата:\n" + selectedFormat);
            } catch (Exception ex) {
                showError("Ошибка чтения файла. Проверьте соответствие формата файла и выбранного ТД!");
            }
        });

        // Кнопка: Очистить список
        clearButton.addActionListener(e -> {
            currentList.clear();
            updateUIList();
        });
    }

    /**
     * Метод для обновления JList на основе данных из CustomList через forEach-итератор
     */
    private void updateUIList() {
        listModel.clear();
        final int[] counter = {0};
        
// Используем наш лямбда-итератор forEach, заложенный в структуре данныхcurrentList.forEach(item -> {String formattedString = String.format("[%d] %s", counter[0], currentBuilder.toString(item));listModel.addElement(formattedString);counter[0]++;});}private void addSideComponent(JPanel panel, Component comp) {comp.setMaximumSize(new Dimension(220, comp.getPreferredSize().height));panel.add(comp);panel.add(Box.createVerticalStrut(8)); // Фиксированный отступ между элементами UI}private void showError(String msg) {JOptionPane.showMessageDialog(this, msg, "Ошибка", JOptionPane.ERROR_MESSAGE);}}

### Шаг 2: Обновление точки входа `Main.java`

Чтобы приложение теперь запускалось не в консоли, а открывало написанное окно, измените ваш запускающий класс следующим образом:

```java
package ru.nstu.app;

import javax.swing.SwingUtilities;
import ru.nstu.app.view.MainFrame;

public class Main {
    public static void main(String[] args) {
        // Запуск Swing-интерфейса в потоке обработки событий (EDT)
        SwingUtilities.invokeLater(() -> {
            MainFrame frame = new MainFrame();
            frame.setVisible(true);
        });
    }
}
